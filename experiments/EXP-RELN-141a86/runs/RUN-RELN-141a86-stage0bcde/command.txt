# Z/N interval positive control (own exact enumeration)
python3 -c "
import json, count_vectors as cv
rows = []
for bits, N, B in [(14, 16339, 47), (16, 65599, 74), (18, 262147, 117)]:
    D = list(range(1, B+1))
    counts3 = cv.count_vector_direct(D, N, 3)
    stats3 = cv.stats_from_count_vector(counts3, N)
    delta3 = cv.dispersion_delta(stats3['mean'], stats3['concentration'])
    E3 = N*(stats3['concentration']+stats3['mean'])
    rows.append({'bits':bits,'N':N,'B':B,'delta_measured':delta3,'E3_measured':E3})
json.dump(rows, open('runs/RUN-RELN-141a86-stage0bcde/control-tables/zn-interval-positive-control.json','w'), indent=2)
"
